updateItem: (index, field, value) => {
  set(state => {
    const newItems = [...state.items];
    const item = newItems[index];
    if (!item) return state;
    switch (field) {
      case 'quantity': item.quantity = Number(value) || 0; break;
      case 'price': item.price = Number(value) || 0; break;
